cmake_minimum_required(VERSION 3.21)
project(NetcodeExamples LANGUAGES C CXX)
include(CTest)
find_package(NetcodeCore 0.1.0 EXACT CONFIG REQUIRED)
option(NETCODE_COCOS "Build the Cocos yard and gameplay examples" OFF)
option(NETCODE_COCOS_SERVER "Build the dedicated Cocos yard" OFF)
option(NETCODE_GODOT "Build the Godot yard" OFF)
option(NETCODE_UNITY "Build the Unity yard's optional native bridge" OFF)
option(NETCODE_STEAM "Build Steam-enabled examples with a Steam-enabled Core SDK" OFF)
add_subdirectory(puzzle)
add_subdirectory(yard/shared)
if(NETCODE_COCOS OR NETCODE_COCOS_SERVER)
    add_subdirectory(yard/cocos2dx)
endif()
if(NETCODE_GODOT)
    add_subdirectory(yard/godot-native)
endif()
if(NETCODE_UNITY)
    add_subdirectory(yard/unity-native)
endif()
if(BUILD_TESTING)
    foreach(case IN ITEMS match prediction)
        add_executable(${case}_test tests/${case}_test.cpp)
        target_link_libraries(${case}_test PRIVATE yard_match)
        add_test(NAME ${case} COMMAND ${case}_test "${PROJECT_SOURCE_DIR}/yard/shared/replicators.cfg")
    endforeach()
endif()
