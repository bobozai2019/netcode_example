        set(COCOS_SERVER_EXTERNAL "${NETCODE_COCOS_EXTERNAL}")

    if(NOT EXISTS "${NETCODE_COCOS_EXTERNAL}/tinyxml2/tinyxml2.cpp")
        message(FATAL_ERROR "Set NETCODE_COCOS_EXTERNAL to the extracted metal-support-22 source dependencies.")
    endif()

    list(TRANSFORM engine_headers REPLACE "^${COCOS2DX_ROOT_PATH}(/|$)" "${COCOS2DX_ROOT}\\1")
    set_property(TARGET cocos2d PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${engine_headers}")

    "${COCOS2DX_ROOT}/cocos/platform"
)

    target_compile_definitions(cocos2d PUBLIC CC_USE_PHYSICS=$<BOOL:${NETCODE_CHIPMUNK}> CC_USE_3D_PHYSICS=$<BOOL:${NETCODE_BULLET}> CC_USE_NAVMESH=0)
    if(WIN32)
        get_target_property(engine_sources cocos2d SOURCES)
        list(FILTER engine_sources EXCLUDE REGEX "^(audio|network)/.*[.]cpp$")
        set_property(TARGET cocos2d PROPERTY SOURCES "${engine_sources}")
        # Use the pointer-sized Win32 subclass indices on x64.
        set_source_files_properties("${COCOS2DX_ROOT}/cocos/ui/UIEditBox/UIEditBoxImpl-win32.cpp"
            TARGET_DIRECTORY cocos2d PROPERTIES
            COMPILE_DEFINITIONS "GWL_WNDPROC=GWLP_WNDPROC;GWL_USERDATA=GWLP_USERDATA")
        # The x64 dependency shim only provides external/. The public headers live in the engine checkout.
if(WIN32 AND NOT NETCODE_COCOS_SERVER)
    get_target_property(engine_headers cocos2d INTERFACE_INCLUDE_DIRECTORIES)
    list(TRANSFORM engine_headers REPLACE "^${COCOS2DX_ROOT_PATH}(/|$)" "${COCOS2DX_ROOT}\\1")
    set_property(TARGET cocos2d PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${engine_headers}")
endif()

    endif()
endif()
