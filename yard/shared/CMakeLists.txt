add_library(yard_match STATIC layout.cpp match.cpp match_prediction.cpp runner.cpp lobby.cpp)
target_include_directories(yard_match PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(yard_match PUBLIC Netcode::Direct)
if(WIN32)
    target_link_libraries(yard_match PRIVATE iphlpapi ws2_32)
endif()

add_executable(netcode-guest guest_main.cpp)
target_link_libraries(netcode-guest PRIVATE yard_match)
set_target_properties(netcode-guest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NETCODE_STEAM)
    target_compile_definitions(yard_match PRIVATE NETCODE_WITH_STEAM)
    target_link_libraries(yard_match PRIVATE Netcode::Steam)
endif()

set(asset_dirs "${CMAKE_CONFIGURATION_TYPES}")
if(NOT asset_dirs)
    set(asset_dirs ".")
endif()
foreach(config IN LISTS asset_dirs)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${config}")
    if(NETCODE_STEAM)
        configure_file("${NETCODE_STEAMWORKS_ROOT}/redistributable_bin/win64/steam_api64.dll"
                       "${CMAKE_BINARY_DIR}/bin/${config}/steam_api64.dll" COPYONLY)
    endif()
    foreach(asset IN ITEMS replicators.cfg beacon.obj)
        configure_file("${asset}" "${CMAKE_BINARY_DIR}/bin/${config}/${asset}" COPYONLY)
    endforeach()
endforeach()
